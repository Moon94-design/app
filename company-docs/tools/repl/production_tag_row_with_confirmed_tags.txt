<div style={{ display: "grid", gridTemplateColumns: "110px 1fr", gap: 10, alignItems: "start" }}>
            <div className="p" style={{ marginTop: 0 }}>태그</div>
            <div>
              {(() => {
                const tags = (draft.tagsText || "")
                  .split(",")
                  .map((x) => x.trim())
                  .filter(Boolean);

                if (tags.length === 0) return null;

                return (
                  <div style={{ display: "grid", gap: 6, marginBottom: 10 }}>
                    <div style={{ fontWeight: 900, fontSize: 12, opacity: 0.7 }}>
                      확정 태그(클릭 삭제)
                    </div>

                    {tags.map((tag) => (
                      <div
                        key={tag}
                        style={{
                          display: "flex",
                          gap: 8,
                          alignItems: "stretch",
                          borderRadius: 10,
                          overflow: "hidden",
                          background: "rgba(0,0,0,0.22)",
                        }}
                      >
                        {/* 3/4 */}
                        <div
                          style={{
                            flex: "0 0 75%",
                            padding: "8px 10px",
                            fontWeight: 900,
                            fontSize: 13,
                            whiteSpace: "nowrap",
                            overflow: "hidden",
                            textOverflow: "ellipsis",
                          }}
                        >
                          #{tag}
                        </div>

                        {/* 1/4 */}
                        <button
                          type="button"
                          className="btn danger"
                          style={{
                            flex: "0 0 25%",
                            padding: "8px 10px",
                            borderRadius: 0,
                            display: "flex",
                            alignItems: "center",
                            justifyContent: "center",
                          }}
                          onMouseDown={(e) => e.preventDefault()}
                          onClick={() => {
                            const next = tags.filter((x) => x !== tag).join(", ");
                            persist({ ...draft, tagsText: next });
                          }}
                        >
                          삭제
                        </button>
                      </div>
                    ))}
                  </div>
                );
              })()}

              <TagInputText
                value={draft.tagsText}
                onChange={(next) => persist({ ...draft, tagsText: next })}
                scope="production"
              />
            </div>
          </div>

          <div className="divider" />
